# 动作空间

    low=-100.0,
    high=100.0,
    shape=(1,),
    dtype=np.float32
)

    """
    Args:
        normalized_action: 网络输出 [-1, 1]
    Returns:
        实际输入值 [-100, 100]
    """
    return normalized_action * 100.0

# 输入特性常量
